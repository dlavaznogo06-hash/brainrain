function createRoulette(elementId, items) { const track = document.getElementById(elementId); let position = 0; let lastTime = null; const itemWidth = 112; const gap = 9; const step = itemWidth + gap; const speed = 65; function addItem(item) { const card = document.createElement("div"); card.className = "roulette-item " + item.rarity; const image = document.createElement("img"); image.src = item.image; image.alt = item.name; const name = document.createElement("span"); name.textContent = item.name; card.appendChild(image); card.appendChild(name); track.appendChild(card); } // Создаём длинную постоянную ленту. for (let i = 0; i < 20; i++) { addItem(items[i % items.length]); } function animate(time) { if (lastTime === null) lastTime = time; const delta = Math.min((time - lastTime) / 1000, 0.05); lastTime = time; position += speed * delta; // Переносим первую карточку в конец ленты. while (position >= step) { track.appendChild(track.firstElementChild); position -= step; } track.style.transform = `translate3d(${-position}px, 0, 0)`; requestAnimationFrame(animate); } requestAnimationFrame(animate); }
